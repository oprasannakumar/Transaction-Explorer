const SS_ID = '18kXdV216Q--8cFt4AkTT3cmwfJ1s9VAT5ok1CwcsyKQ'; function doGet(e) { try { var action = e.parameter.action; var ss = SpreadsheetApp.openById(SS_ID); var results = []; // 1. Fetch Savings from Named Range var savingsVal = 0; try { var range = ss.getRangeByName("Savings"); if (range) savingsVal = range.getValue(); } catch(err) { console.log("Named range 'Savings' not found."); } // 2. Select sheets based on action var sheets = (action === 'trace') ? ss.getSheets() : [ss.getSheets()[0]]; sheets.forEach(function(sheet) { var vals = sheet.getDataRange().getValues(); if (vals.length < 2) return; var headers = vals.shift(); var sName = sheet.getName(); var lastCol = headers.length - 1; for (var r = 0; r < vals.length; r++) { var row = vals[r]; var isMatch = false; if (action === 'trace') { var refIds = e.parameter.refId.split(','); isMatch = (refIds.indexOf(String(row[lastCol])) !== -1); } else { var d = new Date(row[0]); var from = new Date(e.parameter.from); var to = new Date(e.parameter.to); from.setHours(0,0,0,0); to.setHours(23,59,59,999); isMatch = (!isNaN(d.getTime()) && d >= from && d <= to); } if (isMatch) { var rowData = []; headers.forEach(function(h, i) { rowData.push({ k: h.toString(), v: row[i] }); }); results.push({ "SheetSource": sName, "data": rowData }); } } }); // Bundle transactions and savings value together var output = { transactions: results, savings: savingsVal }; return ContentService.createTextOutput(JSON.stringify(output)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({"error": err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ids = data.ids.map(String); var ss = SpreadsheetApp.openById(SS_ID); var count = 0; ss.getSheets().forEach(function(sheet) { var vals = sheet.getDataRange().getValues(); if (vals.length < 2) return; var lastCol = vals[0].length - 1; for (var i = vals.length - 1; i >= 1; i--) { if (ids.indexOf(String(vals[i][lastCol])) !== -1) { sheet.deleteRow(i + 1); count++; } } }); return ContentService.createTextOutput("Deleted " + count).setMimeType(ContentService.MimeType.TEXT); } catch (e) { return ContentService.createTextOutput("Error").setMimeType(ContentService.MimeType.TEXT); } }
